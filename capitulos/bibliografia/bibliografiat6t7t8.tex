\section*{Fuentes sobre tendencias formativas y desarrollo profesional en CaixaBank}
% ================================================================

\begin{enumerate}[label={[\arabic*]}, resume]

  \item \textbf{CaixaBank --- Plan Estratégico 2025--2027} --- Presentación del plan ante inversores

        y prioridades de gestión del talento para el periodo 2025--2027.\\
        \url{https://www.caixabank.com/es/sobre-caixabank/nuestro-plan-estrategico.html}

  \item \textbf{CaixaBank --- Informe Anual 2024} --- Datos corporativos: plantilla consolidada

        \url{https://www.caixabank.com/informacionparaaccionistaseinversores/informacioneconomicofinanciera/informeanual_es.html}

  \item \textbf{CaixaBank --- Sostenibilidad: Personas} --- Sección de la web corporativa con
        información sobre la política de formación, los programas de talento, la diversidad y el
        bienestar de la plantilla. Fuente de referencia para los datos de horas de formación,
        Virtaula y la Escuela de Liderazgo.\\
        \url{https://www.caixabank.com/es/sostenibilidad/personas.html}

  \item \textbf{CaixaBankCareers --- ¿Qué te ofrecemos?} --- Sección del portal de empleo
        corporativo con información sobre el modelo de desarrollo profesional interno, el programa
        de movilidad \textit{moving}, los planes de carrera y las iniciativas formativas disponibles
        para la plantilla.\\
        \url{https://caixabankcareers.com/content/Que-te-ofrecemos/?locale=es_ES}

  \item \textbf{SAP --- CaixaBank \& SAP SuccessFactors} --- Información sobre la implantación de
        SAP SuccessFactors en CaixaBank como plataforma de gestión de RRHH en la nube: gestión del
        aprendizaje, evaluación del desempeño, planificación de la sucesión y People Analytics.\\
        \url{https://www.sap.com/spain/customer-testimonials/caixabank.html}

  \item \textbf{Microsoft --- CaixaBank adopta Microsoft 365 Copilot} --- Anuncio de la integración
        de inteligencia artificial generativa (Microsoft Copilot) en los sistemas de CaixaBank
        para apoyar la productividad de los empleados y la personalización del aprendizaje en
        Virtaula.\\
        \url{https://news.microsoft.com/es-es/2024/caixabank-microsoft-copilot/}

  \item \textbf{CaixaBank --- Nota de prensa: integración con Bankia} --- Información sobre el
        proceso de fusión completado en 2021, la integración de plantillas y la gestión del
        capital humano derivada de la unión de ambas entidades.\\
        \url{https://www.caixabank.com/es/prensa/2021/caixabank-bankia-integracion.html}

  \item \textbf{AEDIPE (Asociación Española de Dirección y Desarrollo de Personas)} ---
        Artículos y recursos sobre outdoor training, gamificación en RRHH y tendencias en
        formación corporativa en el sector financiero español.\\
        \url{https://www.aedipe.es/}

  \item \textbf{Whitmore, John} --- \textit{Coaching for Performance: GROWing Human Potential and
        Purpose} (5ª ed.). Nicholas Brealey Publishing, Londres, 2017. Referencia de base para el
        método GROW y el concepto de shadow coaching aplicados en los programas de desarrollo
        directivo de CaixaBank.

  \item \textbf{Harvard Business Review} --- ``Why Reverse Mentoring Works and How to Do It
        Right''. HBR, octubre de 2019. Fundamento teórico del mentoring inverso y su aplicación
        para acelerar la adopción tecnológica en organizaciones con plantillas multigeneracionales.\\
        \url{https://hbr.org/2019/10/why-reverse-mentoring-works-and-how-to-do-it-right}

  \item \textbf{Llorente \& Cuenca / Equipos \& Talento} --- ``La gamificación en la formación
        corporativa: casos de éxito en el sector bancario español''. Artículo de referencia sobre
        el uso de mecánicas de juego en plataformas LMS, incluyendo insignias digitales, retos
        y marcadores de progreso como los implementados en Virtaula.\\
        \url{https://www.equiposytalento.com/}

  \item \textbf{Muñiz, R. / Marketing XXI} --- ``El outdoor training como herramienta de
        desarrollo directivo''. Fuente de referencia para la descripción metodológica del outdoor
        training: diseño de pruebas, grupos de 15--20 personas, sesión reflexiva final y plan
        de cambio.\\
        \url{https://www.marketing-xxi.com/outdoor-training.html}

\end{enumerate}
